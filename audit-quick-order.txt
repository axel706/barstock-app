      const vendor = (r[cols.vendor] && String(r[cols.vendor]).trim()) || inferVendor(code);
      return {
        code:String(code||'').trim(),
        item:String(item||'').trim(),
        itemNorm: normalizeName(item),
        onHand,
        suggested,
        vendor,
        value: 0,
        toOrder: computeToOrder(onHand,suggested),
        orderOverride:''
      };
    })
    .filter(r => r.itemNorm);
}
function fmt(v){ return Number.isInteger(v) ? String(v) : Number(v).toFixed(2).replace(/\.00$/,''); }
function fmtMoney(v){ return '$' + parseNum(v).toLocaleString(undefined,{minimumFractionDigits:2,maximumFractionDigits:2}); }
function escapeHtml(s){return String(s??'').replace(/[&<>"']/g,m=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m]));}

function rowOrderHistoryMatchesItem(orderItem, row){
  const codeA = String(orderItem?.code || '').trim();
  const codeB = String(row?.code || '').trim();
  if(codeA && codeB && codeA === codeB) return true;
  return normalizeName(orderItem?.item || '') === normalizeName(row?.item || '');
}
function getRecentVendorOrders(vendor, limit = 6){
  return (state.orderHistory || [])
    .filter(order => String(order.vendor || '').toUpperCase() === String(vendor || '').toUpperCase())
    .sort((a,b) => new Date(b.createdAt || 0) - new Date(a.createdAt || 0))
    .slice(0, limit);
}
function getRecentRowQuantities(row, limit = 6){
  const orders = getRecentVendorOrders(row?.vendor, limit);
  const quantities = [];
  orders.forEach(order => {
    const match = (order.items || []).find(item => rowOrderHistoryMatchesItem(item, row));
    if(match && parseNum(match.finalOrder) > 0){
      quantities.push(parseNum(match.finalOrder));
    }
  });
  return quantities;
}
function buildRowSuggestion(row){
  const suggested = parseNum(row?.suggested);
  const recentQty = getRecentRowQuantities(row, 6);
  const recentQuick = getRecentQuickOrders(row, 6);
  if(recentQuick.length >= 2){
    const avgQuick = recentQuick.reduce((sum, qty) => sum + qty, 0) / recentQuick.length;
    const recommended = Math.ceil((suggested + avgQuick) / 2);
    if(avgQuick >= 2){
      return { type:'low', label:`Quick orders +${fmt(avgQuick)} avg`, detail:`Consider ${Math.max(suggested+1, recommended)}` };
    }
  }
  if(recentQty.length >= 2){
    const avg = recentQty.reduce((sum, qty) => sum + qty, 0) / recentQty.length;
    if(avg >= suggested + 1 && avg >= Math.max(2, suggested * 1.25)){
      return { type:'low', label:'Suggested low' };
    }
    if(suggested >= 2 && avg <= suggested * 0.5){
      return { type:'high', label:'Suggested high' };
    }
  }
  const recentOrders = getRecentVendorOrders(row?.vendor, 3);
  if(recentOrders.length >= 3){
    const presentRecently = recentOrders.some(order => (order.items || []).some(item => rowOrderHistoryMatchesItem(item, row)));
    if(!presentRecently && suggested > 0){
      return { type:'skip', label:'Not in recent orders' };
    }
  }
  return null;
}
function suggestionChipHtml(row){
  const suggestion = buildRowSuggestion(row);
  if(!suggestion) return '';
  const klass = suggestion.type === 'low'
    ? 'suggestion-low'
    : suggestion.type === 'high'
      ? 'suggestion-high'
      : 'suggestion-skip';
  const title = suggestion.detail ? `${suggestion.label} • ${suggestion.detail}` : suggestion.label;
  return `<span class="suggestion-chip ${klass}" title="${escapeHtml(title)}">${escapeHtml(suggestion.label)}</span>`;
}
function renderSuggestionsSummary(rows){
  const wrap = document.getElementById('smartSuggestionsSummary');
  if(!wrap) return;
  const suggestions = (rows || []).map(row => buildRowSuggestion(row)).filter(Boolean);
  const low = suggestions.filter(s => s.type === 'low').length;
  const high = suggestions.filter(s => s.type === 'high').length;
  const skip = suggestions.filter(s => s.type === 'skip').length;
  const parts = [];
  if(low) parts.push(`${low} suggested low`);
  if(high) parts.push(`${high} suggested high`);
  if(skip) parts.push(`${skip} not in recent orders`);
  if(!parts.length){
    wrap.classList.add('hidden');
    wrap.innerHTML = '';
    return;
  }
  wrap.classList.remove('hidden');
  wrap.innerHTML = `<strong>Smart Suggestions</strong> <span class="small">• ${parts.join(' • ')}</span>`;
}

async function saveQuickOrder(){
  const vendor = quickOrderVendorTab;
  const dateValue = String(document.getElementById('quickOrderDateInput')?.value || '').trim();
  const timeValue = String(document.getElementById('quickOrderTimeInput')?.value || '12:00').trim() || '12:00';
  const note = String(document.getElementById('quickOrderNoteInput')?.value || 'Quick Order').trim() || 'Quick Order';
  if(!vendor || !dateValue || !quickOrderDraftItems.length){
    setStatus('Add at least one item and complete vendor/date.');
    return;
  }
  const when = new Date(`${dateValue}T${timeValue}:00`);
  if(Number.isNaN(when.getTime())){
    setStatus('Invalid quick order date or time.');
    return;
  }
  const items = quickOrderDraftItems.map(item => ({
    ...item,
    finalOrder: parseNum(item.qty || item.finalOrder),
    note
  }));
  const totalUnits = items.reduce((sum, item) => sum + parseNum(item.qty || item.finalOrder), 0);
  const subtotal = items.reduce((sum, item) => sum + (parseNum(item.qty || item.finalOrder) * parseNum(item.value)), 0);
  const exportType = vendor === 'LOOP' ? 'loop_csv' : 'vendor_jpg';
  const filename = vendor === 'LOOP' ? 'LOOP upload-ready CSV' : 'Vendor JPG';
  const entry = {
    id: uniqueOrderId(),
    vendor,
    createdAt: when.toISOString(),
    exportType,
    filename,
    totalUnits: clampNonNegative(totalUnits),
    subtotal: clampNonNegative(subtotal),
    items,
    quickOrder: true,
    note
  };
  try{
    if (typeof window.persistOrderToSupabase !== 'function') {
      throw new Error('persistOrderToSupabase unavailable');
    }

    await window.persistOrderToSupabase(entry);

    if (typeof window.loadOrdersFromCloud === 'function') {
      await window.loadOrdersFromCloud();
    } else {
      state.orderHistory = [entry, ...(state.orderHistory || [])];
      saveState();
      render();
    }

    closeQuickOrderModal();
    setStatus('Quick Order saved to Order History and synced to cloud.');
  } catch(err){
    console.error(err);
    alert('Quick Order was NOT saved to cloud.\n\n' + (err.message || String(err)));
    setStatus('Quick Order failed because cloud save failed.');
  }
}
